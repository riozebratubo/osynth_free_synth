# osynth host port — the engine as a plain-CMake static library.
#
# This compiles the *same* component sources the ESP-IDF build compiles, with
# port/host/include ahead of them on the include path so that the esp_*,
# freertos/* and sdkconfig.h the firmware includes resolve to the shims beside
# this file. No DSP source is edited, and none is copied.
#
# The source list is deliberately written out rather than globbed. Globbing
# would silently pull in a new hardware file the day someone adds one, and the
# failure would land at link time in the app. tools/check_host_sources.py diffs
# this list against the idf_component_register() calls in components/*/ and is
# the thing that keeps the two honest -- run it after touching either.
#
# ---------------------------------------------------------------------------
# Staging
#
# OSYNTH_HOST_SOURCES below is what compiles today (plan step 1: the DSP core).
# The components still to come each need a storage shim first, and each is
# listed in OSYNTH_HOST_PENDING with what blocks it. They are not commented-out
# lines in the main list because a reader should be able to tell "not ported
# yet" from "deliberately excluded", and the two are different answers.
# ---------------------------------------------------------------------------

cmake_minimum_required(VERSION 3.16)

# Usable two ways: configured directly (the demo, and the quickest way to test
# the port), or pulled in with add_subdirectory() by app_osyntho when
# OSYNTHO_EMBEDDED is on. project() is only called in the first case -- a
# nested project() inside a parent build re-initialises the toolchain scope for
# no benefit here.
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(osynth_core LANGUAGES C CXX)
    set(OSYNTH_HOST_TOP_LEVEL ON)
else()
    set(OSYNTH_HOST_TOP_LEVEL OFF)
    # Three of the sources are C -- midi.c, miniaudio.c and the generated drum
    # kit blob -- and a parent project need not have enabled it: app_osyntho
    # declares LANGUAGES CXX. Without this CMake silently skips every .c file
    # and the link fails on symbols whose sources were never compiled, which
    # is a considerably less obvious message than this line is.
    enable_language(C)
endif()

# Set on the target rather than the directory, so embedding this in a project
# that builds at a different standard (app_osyntho is C++23) leaves that
# project's own targets alone.
set(CMAKE_C_STANDARD 11)

# The repo root, so the component paths below read the same as they do in the
# firmware build.
get_filename_component(OSYNTH_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(OSYNTH_COMPONENTS "${OSYNTH_ROOT}/components")

# ---------------------------------------------------------------------------
# Generated headers
#
# factory_wavetables.h is generated by a host-side script on the firmware build
# too (components/engines/CMakeLists.txt) -- stdlib-only python, no numpy -- so
# the same custom command works here unchanged. Regenerates only when the
# script changes, exactly as it does there.
# ---------------------------------------------------------------------------
find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(OSYNTH_GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/gen")
file(MAKE_DIRECTORY "${OSYNTH_GEN_DIR}")
set(OSYNTH_WT_HEADER "${OSYNTH_GEN_DIR}/factory_wavetables.h")

add_custom_command(
    OUTPUT "${OSYNTH_WT_HEADER}"
    COMMAND "${Python3_EXECUTABLE}" "${OSYNTH_ROOT}/tools/gen_wavetables.py"
            -o "${OSYNTH_WT_HEADER}"
    DEPENDS "${OSYNTH_ROOT}/tools/gen_wavetables.py"
    COMMENT "Generating factory wavetables (tools/gen_wavetables.py)"
    VERBATIM
)
add_custom_target(osynth_gen_wavetables DEPENDS "${OSYNTH_WT_HEADER}")

# The factory drum kit, embedded the same way the firmware embeds it -- as
# read-only data the kit is parsed out of in place. components/drums/
# CMakeLists.txt uses target_add_binary_data (objcopy) for this; MSVC has no
# way to reach those symbol names, so tools/bin2c.py emits an equivalent C
# array. See the declarations in drum_kit.cpp for the two-line difference this
# makes on the consumer side.
#
# The image is NOT generated here, exactly as it is not in the firmware build:
# converting a WAV pack needs numpy/scipy. It is built once with
#     python tools/gen_drumkit.py --pack <folder> --out tools/out/drumkit.bin
# and with no image present the build still links -- an empty kit is emitted
# and the drum bus starts silent, which is the same fallback the firmware has.
if(NOT DEFINED OSYNTH_DRUMKIT)
    set(OSYNTH_DRUMKIT "${OSYNTH_ROOT}/tools/out/drumkit.bin"
        CACHE FILEPATH "Drum kit image embedded into the host build")
endif()

set(OSYNTH_KIT_BLOB "${OSYNTH_GEN_DIR}/drumkit_blob.c")

if(EXISTS "${OSYNTH_DRUMKIT}")
    file(SIZE "${OSYNTH_DRUMKIT}" _kit_size)
    math(EXPR _kit_kb "${_kit_size} / 1024")
    message(STATUS "osynth host: embedding drum kit ${OSYNTH_DRUMKIT} (${_kit_kb} KB)")
    add_custom_command(
        OUTPUT "${OSYNTH_KIT_BLOB}"
        COMMAND "${Python3_EXECUTABLE}" "${OSYNTH_ROOT}/tools/bin2c.py"
                --in "${OSYNTH_DRUMKIT}" --out "${OSYNTH_KIT_BLOB}"
                --symbol drumkit_bin
        DEPENDS "${OSYNTH_DRUMKIT}" "${OSYNTH_ROOT}/tools/bin2c.py"
        COMMENT "Embedding drum kit (${_kit_kb} KB)"
        VERBATIM
    )
else()
    message(STATUS
        "osynth host: no drum kit at ${OSYNTH_DRUMKIT} -- the drum bus will be "
        "silent. Build one with: python tools/gen_drumkit.py --pack <folder> "
        "--out tools/out/drumkit.bin")
    add_custom_command(
        OUTPUT "${OSYNTH_KIT_BLOB}"
        COMMAND "${Python3_EXECUTABLE}" "${OSYNTH_ROOT}/tools/bin2c.py"
                --empty --out "${OSYNTH_KIT_BLOB}" --symbol drumkit_bin
        DEPENDS "${OSYNTH_ROOT}/tools/bin2c.py"
        COMMENT "Generating an empty drum kit (no image configured)"
        VERBATIM
    )
endif()

# host_compat.h ahead of every translation unit. The GCC spellings the DSP
# uses -- __restrict__ (70 declarations) and __builtin_expect via SYNTH_LIKELY
# -- appear in headers no ordinary include order reaches first, so a normal
# #include cannot cover them. See that header for what it does and does not
# handle.
if(MSVC)
    # One argument, quoted as a whole: MSVC takes the path glued to /FI, and
    # splitting it across two list items makes cl treat the path as a source.
    target_compile_options(osynth_core PRIVATE
        "/FI${CMAKE_CURRENT_LIST_DIR}/include/host_compat.h")
else()
    target_compile_options(osynth_core PRIVATE
        -include "${CMAKE_CURRENT_LIST_DIR}/include/host_compat.h")
endif()

# miniaudio is upstream third-party C: it is not osynth code, it does not use
# the GCC spellings host_compat.h exists for, and it emits warnings we have no
# business "fixing" in a vendored file. Excluded from both.
set_source_files_properties(
    "${CMAKE_CURRENT_LIST_DIR}/third_party/miniaudio.c"
    PROPERTIES COMPILE_OPTIONS ""
)

# SYNTH_TARGET_HOST identifies this build to the firmware sources, and it has
# to come from the build system rather than from a header.
#
# The reason is include order. A source that guards an #include on it -- as
# presets.cpp and loop_store.cpp do, choosing a host path over a LittleFS mount
# or an SD card -- is asking the question BEFORE it has included anything, so a
# definition living in sdkconfig.h is not visible yet and the guard silently
# takes the firmware branch. That is exactly the failure it exists to prevent,
# and it presents as a missing ESP-IDF header rather than as a wrong answer.
#
# A compile definition is visible from the first line of every translation
# unit, which is the property a target selector needs.
target_compile_definitions(osynth_core PUBLIC SYNTH_TARGET_HOST=1)

# MSVC deprecates fopen, strerror and strncpy in favour of its own _s variants.
# The firmware uses the standard ones, correctly, on four toolchains; the
# warning is advice about a Microsoft extension rather than a defect, and it
# fires ~30 times across the storage components. PRIVATE: this is a statement
# about compiling these sources, not something to impose on a consumer.
if(MSVC)
    target_compile_definitions(osynth_core PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()

# C++20 as a floor, not a pin: a parent at a higher standard (app_osyntho is
# C++23) keeps its own. PUBLIC so a consumer compiling against these headers
# gets at least this much.
#
# Deliberately not relying on the newer parts of it. std::atomic_ref was used
# once and removed -- the Android NDK's libc++ does not have it -- which is a
# reminder that "C++20" is not one thing across five toolchains.
target_compile_features(osynth_core PUBLIC cxx_std_20)

find_package(Threads REQUIRED)
target_link_libraries(osynth_core PUBLIC Threads::Threads)

# ---------------------------------------------------------------------------
# MIDI input (RtMidi)
#
# RtMidi covers WinMM, CoreMIDI and ALSA from one source file. There is no
# Android or iOS backend, so those build with OSYNTH_HOST_MIDI_IN at 0 and
# midi_in_host.cpp compiles to three stubs -- mobile MIDI is USB host or BLE
# MIDI, a different problem rather than a missing flag.
#
# The API is chosen by a define; getting it wrong yields an RtMidi that
# compiles and finds no ports, which is why each is set beside the libraries
# it needs rather than somewhere central.
# ---------------------------------------------------------------------------
if(ANDROID OR IOS)
    set(OSYNTH_MIDI 0)
elseif(WIN32)
    set(OSYNTH_MIDI 1)
    target_compile_definitions(osynth_core PRIVATE __WINDOWS_MM__)
    target_link_libraries(osynth_core PUBLIC winmm)
elseif(APPLE)
    set(OSYNTH_MIDI 1)
    target_compile_definitions(osynth_core PRIVATE __MACOSX_CORE__)
    target_link_libraries(osynth_core PUBLIC "-framework CoreMIDI")
elseif(UNIX)
    # The one spot in the host build that needs a *development* package.
    # miniaudio dlopens ALSA for audio (the backend block below says so), but
    # RtMidi includes <alsa/asoundlib.h>, so here the headers have to be on
    # disk when it compiles. Naming the package at configure time beats a
    # fatal error a hundred files into the build.
    find_package(ALSA)
    if(NOT ALSA_FOUND)
        message(FATAL_ERROR
            "MIDI input needs the ALSA development headers: install "
            "libasound2-dev (Debian/Ubuntu), alsa-lib-devel (Fedora) or "
            "alsa-lib (Arch).")
    endif()
    set(OSYNTH_MIDI 1)
    target_compile_definitions(osynth_core PRIVATE __LINUX_ALSA__)
    target_include_directories(osynth_core PRIVATE ${ALSA_INCLUDE_DIRS})
    target_link_libraries(osynth_core PUBLIC ${ALSA_LIBRARIES})
else()
    set(OSYNTH_MIDI 0)
endif()

# PUBLIC: osynth_host_midi.h tests it, and a consumer including that header
# has to see the same answer this library was built with.
target_compile_definitions(osynth_core PUBLIC OSYNTH_HOST_MIDI_IN=${OSYNTH_MIDI})

if(OSYNTH_MIDI)
    target_sources(osynth_core PRIVATE
        "${CMAKE_CURRENT_LIST_DIR}/third_party/RtMidi.cpp")
    # Vendored upstream C++: not osynth code, and its warnings are not ours to
    # fix in a file we keep byte-identical.
    set_source_files_properties(
        "${CMAKE_CURRENT_LIST_DIR}/third_party/RtMidi.cpp"
        PROPERTIES COMPILE_OPTIONS "")
endif()

# What miniaudio needs from each platform. Every clause below is from its own
# build notes (miniaudio.h, "2. Building"), not from guesswork -- the failures
# they prevent are a link error at best and a rejected app at worst.
#
# ANDROID is tested before UNIX because CMake sets both on Android, and IOS
# before the APPLE clause would catch it, because APPLE is true on iOS too.
if(WIN32)
    # ole32/oleaut32 for WASAPI's COM interfaces, user32 for the message loop
    # it uses to notice default-device changes.
    target_link_libraries(osynth_core PUBLIC ole32 oleaut32 user32)
elseif(APPLE)
    # MA_NO_RUNTIME_LINKING and explicit frameworks, rather than the default
    # of resolving them with dlopen at run time.
    #
    # On macOS that default is what fails Apple's *notarization*: an app that
    # loads frameworks dynamically needs the allow-dyld-environment-variables
    # and allow-unsigned-executable-memory entitlements to be accepted, which
    # is a much worse trade than naming three frameworks here. miniaudio's own
    # notes say so (see its issue 203).
    #
    # On iOS runtime linking is already off inside miniaudio, so this only
    # makes the two Apple targets agree.
    target_compile_definitions(osynth_core PRIVATE MA_NO_RUNTIME_LINKING)
    target_link_libraries(osynth_core PUBLIC
        "-framework CoreFoundation" "-framework CoreAudio"
        "-framework AudioToolbox")
    if(IOS)
        # iOS drives an AVAudioSession, so miniaudio includes
        # <AVFoundation/AVFoundation.h> -- an Objective-C header. The .c file
        # therefore has to be compiled as Objective-C, which is what its notes
        # mean by "needs to be compiled as Objective-C"; without this the
        # build fails inside a vendored file with errors that look nothing
        # like their cause.
        target_link_libraries(osynth_core PUBLIC "-framework AVFoundation")
        set_source_files_properties(
            "${CMAKE_CURRENT_LIST_DIR}/third_party/miniaudio.c"
            PROPERTIES COMPILE_OPTIONS "-x;objective-c")
    endif()
elseif(ANDROID)
    # Nothing to link. AAudio is the preferred backend and is resolved at run
    # time, with OpenSL|ES as the fallback for Android 7 and older -- also
    # resolved at run time. Naming -lOpenSLES here is only needed alongside
    # MA_NO_RUNTIME_LINKING, and adding it otherwise makes a hard dependency
    # out of a library miniaudio is prepared to find missing.
elseif(UNIX)
    # ALSA and PulseAudio are dlopened, so no development packages are needed;
    # libdl and libm are the whole build-time requirement.
    target_link_libraries(osynth_core PUBLIC ${CMAKE_DL_LIBS} m)
endif()

# No libatomic clause here, and that is a measurement rather than an omission.
# miniaudio's notes say 32-bit ARM "may need -latomic", so both Android ABIs
# were checked with llvm-nm --undefined-only: neither armeabi-v7a nor
# arm64-v8a leaves a single __atomic_* reference, because Clang inlines them
# all. Linking a library nothing calls would be insurance against a
# hypothetical, and would break on any platform that ships without it.

# ---------------------------------------------------------------------------
# osynth_host_demo -- the first-sound gate (plan step 2)
#
# Brings the engine up against a real device and plays notes. Deliberately a
# separate target from the library: the library is what the app links, and
# nothing that ships should carry a main() that plays a scale.
#
# It links today because the chain it runs -- voices then FX -- needs none of
# the components still to be ported. That is also why it is a demo and not the
# host entry point; see the header comment in src/host_demo.cpp.
# ---------------------------------------------------------------------------
# Defaults to the top-level case only: an app embedding the engine has no use
# for a binary that plays a scale and exits.
option(OSYNTH_HOST_DEMO "Build the first-sound demo executable"
       ${OSYNTH_HOST_TOP_LEVEL})

if(OSYNTH_HOST_DEMO)
    add_executable(osynth_host_demo
        "${CMAKE_CURRENT_LIST_DIR}/src/host_demo.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/src/host_prototest.cpp")
    target_link_libraries(osynth_host_demo PRIVATE osynth_core)
    target_include_directories(osynth_host_demo PRIVATE
        "${OSYNTH_COMPONENTS}/audio_io")
    if(MSVC)
        target_compile_options(osynth_host_demo PRIVATE
            "/FI${CMAKE_CURRENT_LIST_DIR}/include/host_compat.h"
            /fp:precise /wd4244 /wd4267)
    else()
        target_compile_options(osynth_host_demo PRIVATE
            -include "${CMAKE_CURRENT_LIST_DIR}/include/host_compat.h"
            -fno-finite-math-only)
    endif()
endif()

# -fno-finite-math-only: the DSP relies on ordinary IEEE comparisons, and the
# firmware is built without fast-math. Stated rather than assumed because a
# parent project that turns fast-math on globally would otherwise change how
# the filters behave here and nowhere else.
if(NOT MSVC)
    target_compile_options(osynth_core PRIVATE
        -fno-finite-math-only
        -Wall
        -Wno-unused-function
    )
else()
    # C4244/C4267: the DSP converts freely between float, int and size_t and
    # the firmware compiler does not warn about it either. /fp:precise is the
    # MSVC default but is named here for the same reason as above.
    target_compile_options(osynth_core PRIVATE /fp:precise /wd4244 /wd4267)
endif()
